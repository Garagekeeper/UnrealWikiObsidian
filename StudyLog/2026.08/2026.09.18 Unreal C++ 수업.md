void UEC_StackDamage::Execute_Implementation(
	const FGameplayEffectCustomExecutionParameters& ExecutionParams, 
	FGameplayEffectCustomExecutionOutput& OutExecutionOutput) const
{
	const UAbilitySystemComponent* TargetASC = ExecutionParams.GetTargetAbilitySystemComponent();
	if (!TargetASC) return;



	// 스택 카운트 세기
	int32 StackCnt = 0;
	if (Debuff)
	{
		FGameplayEffectQuery Query;
		Query.EffectDefinition = Debuff;
		// stack이 원래는 이펙트 하나에 숫자만 올라가는데 혹시 모르니까 순회하면서 확인 
		TArray<FActiveGameplayEffectHandle> ActiveHandles =  TargetASC->GetActiveEffects(Query);
		for (const FActiveGameplayEffectHandle& Handle : ActiveHandles)
		{
			if (const FActiveGameplayEffect* ActiveGE = TargetASC->GetActiveGameplayEffect(Handle))
			{
				StackCnt += ActiveGE->Spec.GetStackCount();
			}
		}
	}



	// 공격력과 크리티컬 확률 가져오기
	
	// 태그 확인하는 법
	// 스펙에 앵간한거 다 들어있음
	// 큐, UI등도 있다면 확인할 수 있다.
	const FGameplayEffectSpec& Spec = ExecutionParams.GetOwningSpec();
	FAggregatorEvaluateParameters EvalueationParameters;
	EvalueationParameters.SourceTags = Spec.CapturedSourceTags.GetAggregatedTags();
	EvalueationParameters.TargetTags = Spec.CapturedTargetTags.GetAggregatedTags();

	// 저주 태그가 붙어 있는지 확인하기
	const FGameplayTagContainer* TargetTags = Spec.CapturedTargetTags.GetAggregatedTags();
	TargetTags->HasTag(FGameplayTag::RequestGameplayTag(FName("GAS.Test.Debuff.Curse")));

	// 생성자에서 캡쳐한 공격력 뽑아오기
	float AttackPower = 0.f;
	ExecutionParams.AttemptCalculateCapturedAttributeMagnitude(StackDamageStatics().AttackPowerDef, EvalueationParameters, AttackPower);
	
	// 생성자에서 캡쳐한 치명타 확률 가져오기
	float CriticalChance = 0.f;
	ExecutionParams.AttemptCalculateCapturedAttributeMagnitude(StackDamageStatics().CriticalChanceDef, EvalueationParameters, CriticalChance);

	float CriticalMultiplier = 1.0f;
	if (FMath::FRand() < CriticalChance)
	{
		CriticalMultiplier *= CriticalRate;
	}


	// Stack 부호비트 침범 방지
	StackCnt = FMath::Clamp(StackCnt, 0, 30);
	// 스택마다 2배씩 강해짐
	const float StackMultiplier = static_cast<int>(1 << StackCnt);
	const float FinalDamage = (BaseDamgae + AttackPower) * StackMultiplier * CriticalMultiplier;
	if (FinalDamage > 0.0f)
	{
		// 모디파이어 추가
		// 여러 모디파이어를 넣을 수 있음 현재는 데미지를 변경하는 모디파이어
		OutExecutionOutput.AddOutputModifier(
			FGameplayModifierEvaluatedData(UStatAttributeSet::GetDamageAttribute(), EGameplayModOp::Additive, FinalDamage)
		);
	}


}

말 그대로 이펙트에 스택을 쌓는것 
- Stacking Type
	- No Stacking
	- Stack Per Target
	- Stack Per Source
	
- Stack Limit Count
	- 최대 스택 갯수
	- 
- Stack Duration Refresh Policy
	- 스택이 새로 쌓일 때  Duration을 어떻게 처리할지
	- Refresh On Successful Application : 성공적으로 적용되면 초기화
	- Never Refresh : 초기화 안함
	- Extend Duration : 남은 시간에 Duration만큼 더하기
	- 
- Stack Period Reset Policy
	- 스택이 새로 쌓일때 현재 주기(틱)를 어떻게 할 것이냐
	* Refresh On Successful Application : 성공적으로 적용되면 틱 초기화
	* NeverRest : 틱 초기화 안함.
	
- Stack Expiration Policy
	- 지속시간이 끝났을 때 스택을 없애는 정책
	- Clear Entire Stack : 싹 다 날림
	- Remove Single Stack and Refresh Duration : 하나만 날리고 Dutarion 초기화
	- Refresh Duration : Duration만 초기화 (타이머 같은 느낌)

- Factor in Stack Count 
	- false면 몇개가 쌓여있어도 하나만 적용됨

# 4. Ability
ASC를 가진 액터가 할 수 있는 행동. Ability를 준다는 개념으로 접근해야 함. Ability를 Activate할 때 Cost와 Cooltime을 만족하는지 판단해서 진행함. 코스트는 어트리뷰트를 만들고, Cooldown의 경우에는 쿨타임 태그를 붙여주는 Duration 이펙트를 만들어서 사용

![[스크린샷 2026-09-18 180753.png]]


- `CanActivateAbility()` : 조건 확인  
- `TryActivateAbility()` : 발동 요청
- `CallActivateAbility()`  : `TryActivateAbility`와 실제 구현체인 `ActivateAbility` 사이를 이어주는 엔진 내부 래퍼(Wrapper) 함수
- `ActivateAbility()` : 실제 실행
- `CommitAbility()` : 사용한 Cost 반영 및 쿨다운 적용
- `CancelAbility()` : 모종의 사유로 취소 되었을 때
- `EndAbility()` : 스킬 사용 종료 (취소든, 조건 안되든 실행했든 무조건 실행)

	// 이 어빌리티는 액터별로 인스턴싱됨.
	InstancingPolicy = EGameplayAbilityInstancingPolicy::InstancedPerActor;

void UGameplayAbility_Blink::ActivateAbility(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, const FGameplayAbilityActivationInfo ActivationInfo, const FGameplayEventData* TriggerEventData)
{
	// 코스트와 쿨다운을 적용한 뒤 
	if (!CommitAbility(Handle, ActorInfo, ActivationInfo))
	{
		EndAbility(Handle, ActorInfo, ActivationInfo, true, true);
		return;
	}


	ACharacter* Character = Cast<ACharacter>(ActorInfo->AvatarActor.Get());
	if (!Character)
	{
		// 캐릭터가 아닌 액터가 이 어빌리티를 사용했다.
		EndAbility(Handle, ActorInfo, ActivationInfo, true, true);
		return;
	}


	// 시작 위치와 현재 회전
	const float CurrentLevel = GetAbilityLevel(Handle, ActorInfo);
	const float Distance = BlinkDistance.GetValueAtLevel(CurrentLevel);


	// 목적지 위치 계산
	const FVector StartLoc = Character->GetActorLocation();
	const FRotator CurrentRotation = Character->GetActorRotation();
	const FVector DestLoc = CalculateBlinkDestination(Character, Distance);


	// 마지막 컴포넌트는 끼임 체크를 하는지 안하는지 (true일 경우 체크 안함)
	const bool bTeleported = Character->TeleportTo(DestLoc, CurrentRotation, false, true);
	if (!bTeleported)
	{
		Character->SetActorLocation(DestLoc, false, nullptr, ETeleportType::TeleportPhysics);
	}

	// 성공적인 종료
	EndAbility(Handle, ActorInfo, ActivationInfo, true, false);
}

FVector UGameplayAbility_Blink::CalculateBlinkDestination(const ACharacter* InCharacter, float InDistance) const
{
	if (!InCharacter) return FVector::ZeroVector;
	UWorld* World = InCharacter->GetWorld();
	if (!World) return InCharacter->GetActorLocation();

	// 이론상 목표 위치
	const FVector StartLoc = InCharacter->GetActorLocation();
	const FVector DesireDestination = StartLoc + (InCharacter->GetActorForwardVector() * InDistance);


	// 충돌 체크용 캡슐 만들기
	const UCapsuleComponent* Capsule = InCharacter->GetCapsuleComponent();
	const float Radius = Capsule->GetScaledCapsuleRadius();
	const float HalfHeight = Capsule->GetScaledCapsuleHalfHeight();
	FCollisionShape CapsuleShape = FCollisionShape::MakeCapsule(Radius, HalfHeight);


	// 스윕 체크할 시작 위치와 종료 위치 게산
	const FVector SweepStart = StartLoc + FVector(0.f, 0.f, StepOffset);
	const FVector SweepEnd = DesireDestination + FVector(0.f, 0.f, StepOffset);

	FCollisionQueryParams QueryParams;
	QueryParams.AddIgnoredActor(InCharacter);

	FHitResult SweepHit;
	bool bHit = World->SweepSingleByChannel(
		SweepHit,
		SweepStart,
		SweepEnd,
		FQuat::Identity,
		ECollisionChannel::ECC_Visibility,
		CapsuleShape,
		QueryParams
	);

	FVector FinalDest = DesireDestination;
	if (bHit && SweepHit.bBlockingHit && !SweepHit.bStartPenetrating && SweepHit.Time > 0.005f)
	{
		FinalDest = SweepHit.Location + (SweepHit.ImpactNormal * CollisionOffset);
		FinalDest.Z = StartLoc.Z;
	}

	if (bTraceFloor)
	{
		FHitResult FloorHit;
		const FVector FloorStart = FinalDest + FVector(0.f, 0.f, 50.f);
		const FVector FloorEnd = FinalDest - FVector(.0f, .0f, TraceFloorTraceDistance);
		if (World->LineTraceSingleByChannel(FloorHit, FloorStart, FloorEnd, ECC_Visibility, QueryParams))
		{
			FinalDest.Z = FloorHit.Location.Z + HalfHeight;
		}
	}

	return FinalDest;
}
