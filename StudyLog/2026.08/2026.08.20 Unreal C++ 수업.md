## 인벤토리 열고 닫기기

인벤토리를 열었을때 입력 모드를 `FInputModeGameOnly`로 설정하면 인벤토리 닫기 단축키가 먹통이된다(당연). 그러면 단축키로 어떻게 닫나요? 위젯에서 입력을 받으면 됩니다. 그런데 입력은 포커스를 받은 위젯에서만 받기 때문에 포커스 설정이 잘 되어야합니다.

**주의!** 포커스를 받을 위젯은 `SetIsFocusable(true)`을 반드시 세팅

**주의!** 아래의 코드로 진행하면 인벤토리가 열렸을때 인벤토리 바깥을 클릭하면 포커스가 나가버려서 슬퍼짐 차라리 나중에 popup ui를 담는 스택을 만들어서 처리를 하던지 해야할 듯.

```cpp file:ActionPlayerController.cpp
void AActionPlayerController::OnInventoryOpenClose(bool bInventoryOpen, UUserWidget* InFocusWidet)
{
	//FInputModeGameOnly
	//FInputModeUIOnly
	//FInputModeGameAndUI

	if (bInventoryOpen)
	{
		// 인풋모드 설정과 포커스 설정
		FInputModeUIOnly InputModeUI;
		InputModeUI.SetWidgetToFocus(InFocusWidet->TakeWidget());
		SetInputMode(InputModeUI);
		SetShowMouseCursor(true);
	}
	else
	{
		FInputModeGameOnly InputModeGame;
		SetInputMode(InputModeGame);
		SetShowMouseCursor(false);
	}

}
```

```cpp file:InventoryWidget.cpp hide:3-10,39-44,13-17 hlt:24,30
void UInventoryWidget::InitInventoryWidget(UInventoryComponent* InventoryComponent)
{
	ClearInventoryWidget();
	TargetInventory = InventoryComponent;

	if (TargetInventory.IsValid())
	{
		UE_LOG(LogTemp, Log, TEXT("TargetInventory was InValid Pointer"));
	}

	if (SlotGridPannel)
	{
		int32 ChildCount = SlotGridPannel->GetChildrenCount();
		int32 InvenSize = TargetInventory.Get()->GetSize();
		SlotSize = FMath::Min(ChildCount, InvenSize);

		SlotWidgets.Empty(SlotSize);
		for (int i = 0; i < SlotSize; ++i)
		{
			if (UInventorySlotWidget* SlotWidget = Cast<UInventorySlotWidget>(SlotGridPannel->GetChildAt(i)))
			{
				// 슬롯UI를 초기화할 때, 슬롯에 마우스가 들어오고 나갈때 호출될 델리게이트에 함수 바인딩
				SlotWidget->InitSlot(TargetInventory.Get(), i, this);
				SlotWidget->OnSlotEnter.AddWeakLambda(
					this,
					[this](int32 InIndex)
					{
						DetailInfo->DetailOpen(TargetInventory.Get()->GetSlot(InIndex)->ItemData);
					});
				SlotWidget->OnSlotLeave.AddWeakLambda(
					this,
					[this]()
					{
						DetailInfo->DetailClose();
					});
				SlotWidgets.Add(SlotWidget);
			}
		}
	}

	TargetInventory->OnMoneyChaneged.AddUObject(this, &UInventoryWidget::RefreshMoneyPannel);
	TargetInventory->OnSlotChanged.BindUObject(this, &UInventoryWidget::RefreshSlotWidget);

	RefreshInventoryWidget();
}
```

```cpp file:DetailInfoWidget.cpp
void UDetailInfoWidget::UpdateLocation()
{
	if (!CanvasSlot.IsValid())
	{
		CanvasSlot = Cast<UCanvasPanelSlot>(Slot);
	}

	if (CanvasSlot.IsValid())
	{
		// 현재 모니터에서 마우스의 좌표를 가져옴
		const FVector2D MouseAbsolute = UWidgetLayoutLibrary::GetMousePositionOnPlatform();
		// 부모 패널(Canvas Panel)을 가져옴
		if (UPanelWidget* ParentPanel = GetParent())
		{
			// 부모 패널의 위치를 기준으로 모니터에서의 마우스 좌표를 가져옴
			const FGeometry& ParentGeometry = ParentPanel->GetTickSpaceGeometry();
			const FVector2D LocalPos = ParentGeometry.AbsoluteToLocal(MouseAbsolute);
			// 가져온 좌표에 디테일패널 위치 시키기
			CanvasSlot->SetPosition(FVector2D(LocalPos));
		}
	}
}

GetMousePositionOnPlatform()  : 모니터 전체를 기준으로한 절대적인 좌표
GetMousePositionOnViewport()  : 뷰포트 위젯의 로컬 공간기준 좌표계, 원본 픽셀 좌표계
GetMousePositionScaledByDPI() : UI크기가 해상도나 DPI에 의해 조절될때 변환된 좌표를 반환
```

# 2. 드래그 드롭
드래그 드롭을 위해서는 `NativeOnMouseButtonDown`, `NativeOnDragCancelled`, `NativeOnDrop`, `NativeOnDragDetected` 이렇게 4개의 함수가 필수적으로 필요하다. 

```cpp file:InventorySlotWidget.cpp

// 에셋이 포커스를 가지고 있을때 마우스 입력이 들어오면 호출되는 함수
FReply UInventorySlotWidget::NativeOnMouseButtonDown(const FGeometry& InGeometry, const FPointerEvent& InMouseEvent)
{
	// 좌클릭인 경우
	if (InMouseEvent.IsMouseButtonDown(EKeys::LeftMouseButton))
	{
		//  해당 슬롯의 정보를 가져온다.
		if (FInventorySlot* InvenSlot = TargetInventory->GetSlot(Index))
		{
			// 빈 슬롯인 경우 부모호출 후 종료
			if (InvenSlot->IsEmpty())
				return Super::NativeOnMouseButtonDown(InGeometry, InMouseEvent);

			// 빈 슬롯이 아니면 현재 입력은 consume하고 Slate/UMG 시스템에게 드래그를 탐지하도록 알려줌
			// 클릭을 받은 위젯, 드래그를 적용할 키를 순차적으로 전단
			return FReply::Handled().DetectDrag(TakeWidget(), EKeys::LeftMouseButton);
		}
	}
	
	// 나머지의 경우 부모를 호출하고 종료
	return Super::NativeOnMouseButtonDown(InGeometry, InMouseEvent);
}

// 드래그를 감지했을때 호출되는 함수
void UInventorySlotWidget::NativeOnDragDetected(const FGeometry& InGeometry, const FPointerEvent& InMouseEvent, UDragDropOperation*& OutOperation)
{
	Super::NativeOnDragDetected(InGeometry, InMouseEvent, OutOperation);

	// 드래그 이벤트에 전달될 UDragDropOperation의 파생 클래스를 생성
	UInventoryDragDropOperation* DragOp = NewObject<UInventoryDragDropOperation>();
	// 드래그동안 보일 위젯을 제작
	UTemporaryWidget* DragTemp = CreateWidget<UTemporaryWidget>(
		this,
		TargetInventory->GetTempSlotWidgetClass()
	);
	
	// 드래그중에 보일 위젯을 방금 만든 위젯으로 설정
	DragOp->DefaultDragVisual = DragTemp;
	// 추가로 전달하고 싶은 속성을 변수 형태로 전달
	DragOp->Index = Index;
	OutOperation = DragOp;
	
	// 현재 슬롯의 아이템의 정보를 토대로 만들어진 임시 위젯을 초기화
	FInventorySlot* InvenSlot = TargetInventory->GetSlot(Index);
	DragTemp->InitTemporarySlotWidget(InvenSlot->ItemData, InvenSlot->GetCnt());

	// 현재 슬롯의 아이템을 임시 슬롯으로 Move
	FCommandResult Res;
	TargetInventory->ExecuteCommand(FInventoryCommand::MakeMove(Index, TargetInventory->GetSize()), Res);

}

// 드래그하던 키에서 손을 떼면(드래그가 종료되면, 드랍되면) 호출되는 함수 
// 여기서의 반환값을 기반으로 실패여부 판단
// 마우스 좌클릭을뗀 위치 아래에서 NativeOnDrop을 실행 Super::NativeOnDrop은 기본적으로 false를 반환하는듯
// 실행 주체는 마우스 밑에 있던 위젯
bool UInventorySlotWidget::NativeOnDrop(const FGeometry& InGeometry, const FDragDropEvent& InDragDropEvent, UDragDropOperation* InOperation)
{
	if (UInventoryDragDropOperation* DragOp = Cast<UInventoryDragDropOperation>(InOperation))
	{
		// 임시슬롯의 아이템을 해당 슬롯으로 이동(Swap)
		// 해당 슬롯의 내용을 드래그가 시작된 슬롯으로 이동
		FCommandResult Res;
		TargetInventory->ExecuteCommand(FInventoryCommand::MakeMove(TargetInventory->GetSize(), Index), Res);
		TargetInventory->ExecuteCommand(FInventoryCommand::MakeMove(TargetInventory->GetSize(), DragOp->Index), Res);
		return true;
	}
	return false;
}


// 드롭이 실패하면 호출되는 함수 
// NativeOnDrop이 false가 되면 이쪽으로 들어오는 듯
void UInventorySlotWidget::NativeOnDragCancelled(const FDragDropEvent& InDragDropEvent, UDragDropOperation* InOperation)
{
	if (UInventoryDragDropOperation* DragOp = Cast<UInventoryDragDropOperation>(InOperation))
	{
		FCommandResult Res;
		// 드롭이 실패한 곳의 좌표를 가져옴
		const FVector2D DropScreenPos = InDragDropEvent.GetScreenSpacePosition();
		// 드롭 실패시 원복 혹은 버리기를 구분할 경계를 위한 부모 위젯
		// 부모 위젯 없으면 원복
		if (!ParentInven.IsValid())
		{
			TargetInventory->ExecuteCommand(FInventoryCommand::MakeMove(TargetInventory->GetSize(), DragOp->Index), Res);
		}
		else
		{
			// 부모 위젯의 Geometry를가져옴
			const FGeometry& InvenGeometry = ParentInven->GetTickSpaceGeometry();

			// 드랍 실패지점의 좌표가 Geometry안에 포함되면 
			if (InvenGeometry.IsUnderLocation(DropScreenPos))
			{
				// 원복
				TargetInventory->ExecuteCommand(FInventoryCommand::MakeMove(TargetInventory->GetSize(), DragOp->Index), Res);
			}
			else
			{
				// 아이템 버리기
				FVector Loc = GetOwningPlayerPawn()->GetActorLocation();
				TargetInventory->ExecuteCommand(FInventoryCommand::MakeDrop(TargetInventory->GetSize(), Loc), Res);
			}
		}

	}

	return Super::NativeOnDragCancelled(InDragDropEvent, InOperation);
}
```
