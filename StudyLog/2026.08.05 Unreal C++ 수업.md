## 일반화된 오브젝트 풀 만들기
키를 `TSubClassOf<AActor>` 와 `UClass*`로 할때의 차이?
사실 `TSubClassOf<AActor>` 자체가 `UClass`를 래핑하는 함수라 어떤걸 써도 동일하게 동작한다. 하지만, 컴파일 타임의 타입 안정성을 위해서 `TSubClassOf<AActor>`을 쓰도록 하자.

### 데이터 에셋을 기반으로 풀 구조체 생성하기

데이터 에셋을 통해서 풀링 정책을 정의했다면 이제 실제 사용을 할 차례다. 저번 시간에는 하나의 액터 타입을 위한 풀을 만들었다. 이번에는 여러 액터가 사용할 수 있도록 풀 서브시스템을 수정한다. 먼저 Pool의 역할을 하는 ReadyActors, ActiveActors를 구조체로 분리해서 타입마다 자신의 고유한 풀을 가질 수 있도록 설계했다.

설계를 알맞게 했다면 이제 설계를 통해서 실제 풀을 생성한다. 서브 시스템의 초기화 단계에서 데이터 에셋을담은 디벨로퍼 세팅을 불러온다. 디벨로퍼 세팅에는 앞서 말한 데이터에셋을 담은 배열이 들어있는데 내부의 데이터 에셋은 `TSoftObjecPtr`타입 이므로 명시적으로 로딩을 하고 넘어간다. 로딩한 에셋을 기반으로 풀을 생성하면서 초기 크기, 최대 크기, 정책등을 정해준다.

```cpp file:ObjectPoolSubSystem.cpp
void UObjectPoolSubsystem::Initialize(FSubsystemCollectionBase& Collection)
{
	Super::Initialize(Collection);

	const UObjectPoolSettings* Settings = GetDefault<UObjectPoolSettings>();
	if (Settings)
	{
		//Settings 내부의 데이터에셋 Tarray를 순회
		for (const TSoftObjectPtr<UObjectPoolDataAsset>& DataAsset : Settings->PoolableDataAsset)
		{
			// 데이터 에셋이 없으면 건너뜀
			if (DataAsset.IsNull()) continue;
			
			// 데이터 에셋을 로딩(SoftObjectPtr)
			TObjectPtr<UObjectPoolDataAsset> LoadedDataAsset = DataAsset.LoadSynchronous();
			// 데이터 에셋을 기반으로 풀 생성
			CreatePool(LoadedDataAsset);
			//PoolMap.FindOrAdd(LoadedDataAsset->ActorClass.LoadSynchronous());
		}
	}
}

void UObjectPoolSubsystem::CreatePool(const UObjectPoolDataAsset* LoadedDataAsset)
{

	// 데이터 에셋 내부에서 풀링할 타입을 불러옴
	TSubclassOf<AActor> InClass = LoadedDataAsset->ActorClass.LoadSynchronous();
	
	// 해당 타입의 풀이 이미 있으면, 지우고 시작
	if (PoolMap.Find(InClass))
	{
		ClearPool(InClass);
	}

	// 새로운 풀을 만들고, 맵에 등록
	PoolMap.FindOrAdd(InClass, 
		FActorPool(LoadedDataAsset->InitialSize, LoadedDataAsset->MaxSize, LoadedDataAsset->PoolPolicy));

}
```


### 풀을 사용한 Spawn

풀의 로직은 지난 코드와 크게 다르지는 않지만 호출자가 소환할 액터를 전달하는 방식으로 변경 되었다. 소환할 액터를 전달하면 ReadyActor 즉 Pool에 가용한 자원이 있으면 Pool에서 받아오고 그렇지 않은 경우는 새롭게 생성하는게 저번시간의 방식이었다. 이번에는 정의한 풀링 정책을 기반으로 새롭게 액터를 생성한다. Pool의 크기가 설정한 최대 사이즈보다 작다면 그냥 새로 만들면 된다. 아래에서 이야기하는 정책들은 풀이 꽉 찼을때를 의미한다.

먼저 ReuseOldest의 경우 가장 오래된 액터를 재사용하는 정책이다. 이를 위해서는 가장 오래된 액터가 어떤 것인지 구분할 기준이 필요했다. 다른 여러 방법들이 있겠지만 가장 먼저 떠오른 방법은 액터를 Pool에서 받아올 때 일종의 TimeStamp를 찍어서 기록하는 것이었다. 물론 풀의 사이즈가 커지고 액터들이 많아지면 가장 오래된 TimeStamp를 찾는데 많은 시간을 쓰겠지만 제일 간단하기도 하고 무엇보다 다른 방법은 떠오르지 않았다. (0806 수업을 듣고 다른 방법이 있는지 알아보자) 현재 사용중인 액터들을 순회하며 타임 스탬프를 보고 가장 오래된 타임 스탬프를 가진 액터를 소환한 액터로 간주하고 위치만 초기화 해주면 마무리된다.

DoNotSpawn은 말그대로 소환하지 않는 정책이다. 자원이 한정적인 환경에서 사용할 듯 하다.

ForceGrow은 최대크기는 상관하지 않고 요청이 들어오는대로 계속 만들어준다. 저번 시간에 사용한 방식이다.

```cpp file:ObjectPoolSubSystem.cpp

// InType 타입의 액터를 소환시도하는 함수
AActor* UObjectPoolSubsystem::Spawn(TSubclassOf<AActor> InType, const FTransform& InTransForm)
{
	if (!InType) return nullptr;
	if (!GetWorld()) return nullptr;
	if (!PoolMap.Find(InType))
	{
		UE_LOG(LogTemp, Log, TEXT("This class can not ppoolable"));
		return nullptr;
	}

	AActor* Spawned = nullptr;

	// 풀에 대기중인 액터가 있으면(풀에 가용한 액터가 있으면)
	if (PoolMap[InType].ReadyActors.Num() > 0)
	{
		// 풀에서 받아와서 반환해준다
		Spawned = PoolMap[InType].ReadyActors.Pop();
		Spawned->SetActorTransform(InTransForm, false, nullptr, ETeleportType::TeleportPhysics);

	}
	// 풀에 대기중인 액터가 없으며면!
	else
	{
		// 정책에 따라서 분기한다.
		switch (PoolMap[InType].Policy)
		{
			// 재사용 정책
			case EObjectPoolPolicy::ReUseOldest:
				// 현재 사용중인 액터가 설정된 최대 크기보다 같거나 크면
				if (PoolMap[InType].ActiveActors.Num() >= PoolMap[InType].MaxSize)
				{
					// 가장 오래된 액터를 찾는다
					// 타임 스탬프 기반 탐색
					TObjectPtr<AActor> Oldest = nullptr;
					double MinVal = GetWorld()->GetTimeSeconds();
					
					// 해당 타입의 풀을 순회
					for (auto& [Key, Val] : PoolMap[InType].ActiveActors)
					{
						if (Val < MinVal)
						{
							Oldest = Key;
							MinVal = Val;
						}
					}

					// 가장 오래된 액터를 반환
					Spawned = Oldest;
					Spawned->SetActorTransform(InTransForm, false, nullptr, ETeleportType::TeleportPhysics);
					UE_LOG(LogTemp, Log, TEXT("%s, Reused by ReUseOldest"), *(InType->GetName()));
				}
				else
				{
					// 풀에 대기중인 액터는 없지만 사이즈가 설정한 최댓값보다 작은 경우
					// 그냥 소환한다.
					SpawnActorToWorld(InType, Spawned, InTransForm);
				}
				break;

				//소환하지 않는 정책
			case EObjectPoolPolicy::DoNotSpawn:
				if (PoolMap[InType].ActiveActors.Num() < PoolMap[InType].MaxSize)
				{
					SpawnActorToWorld(InType, Spawned, InTransForm);
				}
				else
				{
					UE_LOG(LogTemp, Log, TEXT("%s, Not Spawned by DoNotSpawn"), *(InType->GetName()));
				}

				break;
				
				// 상관없이 늘리는 정책책
			case EObjectPoolPolicy::ForceGrow:
				if (PoolMap[Target].ActiveActors.Num() >= PoolMap[InType].MaxSize)
					UE_LOG(LogTemp, Log, TEXT("%s, Spawned by ForceGrow"), *(InType->GetName()));
				SpawnActorToWorld(InType, Spawned, InTransForm);
				break;
			default:
				break;
		}

	}

	// 소환된 액터(진짜 생성, 풀에서 받아옴, 재활용 함 등)가 있으면
	if (Spawned)
	{
		// 해당 액터가 풀러블 인터페이스를 상속 받았는지 확인하고
		// 소환시 이벤트 호출
		if (Spawned->GetClass()->ImplementsInterface(UPoolable::StaticClass()))
		{
			IPoolable::Execute_Onspawn(Spawned);
		}

		else
		{
			// 일반 액터의 경우 그냥 화면에만 안보이게
			Spawned->SetActorHiddenInGame(false);
			Spawned->SetActorTickEnabled(true);
			Spawned->SetActorEnableCollision(true);
		}

		// 해당 타입의 풀에 등록
		PoolMap[InType].ActiveActors.Add(Spawned, GetWorld()->GetTimeSeconds());
	}

	return Spawned;
}


// 사용이 끝난 액터를 풀에 반환
void UObjectPoolSubsystem::ReturnPool(AActor* InActor)
{
	if (!InActor) return;
	UClass* Target = InActor->GetClass();

	if (!PoolMap.Find(Target)) return;
	if (InActor->GetClass()->ImplementsInterface(UPoolable::StaticClass()))
	{
		IPoolable::Execute_OnReturn(InActor);
	}
	else
	{
		InActor->SetActorHiddenInGame(true);
		InActor->SetActorTickEnabled(false);
		InActor->SetActorEnableCollision(false);
	}

	// 사용중인 액터 목록에서 지우고 대기중에 액터 목록으로 이동
	PoolMap[Target].ActiveActors.Remove(InActor);
	PoolMap[Target].ReadyActors.Add(InActor);

}
```


### Warmup

오브젝트 풀을 사용하면서 미리 풀에 액터를 채워놓지 않는다면 오브젝트 풀의 장점을 반만 사용하는 것이다. 오브젝트 풀은 반복적인 소환 삭제를 줄여서 메모리 파편화 혹은 성능의 개선을 기대하는데 미리 채워놓지 않으면 어느정도 풀이 채워지기 전까지는 풀의 역할을 하지 못한다. 탄막 게임을 예시로 들면 로딩 시에 액터를 미리 100개 풀에 채워 넣으면 로딩이 조금 오래 걸릴 지언정 게임에서 풀의 장점을 적극 활용하게 된다. 반면 풀을 미리 채워놓지 않는다면 적절한 액터의 개수가 되기까지 계속해서 새로 생성하게 되고 인게임 상에서 프레임 드랍의 원인이 될 수 있다. 게이머 입장에서 봐도 게임에서 프레임 드랍이 발생하는 것 보다는 로딩이 조금 더 걸리는 편이 낫다.
우리가 만든 서브 시스템에서는 Initailize할때 Warmup을 할 수 없다는 것에 주의해야 한다. `ObjectPoolSubSystem`은 게임 인스턴스의 서브시스템이기 때문에 Initailize시점에 World가 없다. Actor를 소환하려면 World가 있어야하기 때문에 이 타이밍에는 할 수 없고 게임모드의 `BeginPlay()` 등에서 실행하자.

```cpp file:ObjectPoolSubSystem.cpp

void UObjectPoolSubsystem::WarmUpAll()
{
	for (auto& [Key, _] : PoolMap)
	{
		WarmUp(Key);
	}
}

// 풀에 액터를 미리 채워주는 함수
void UObjectPoolSubsystem::WarmUp(TSubclassOf<AActor> InClass)
{
	if (FActorPool* Pool = PoolMap.Find(InClass))
	{
		// 눈에 보이지 않는 곳으로 소환
		FTransform Init(FVector::DownVector * 10000.0f);
		TArray<AActor*> SpawnedArray;
		// 초기 사이즈 만큼 생성
		for (int i = 0; i < Pool->InitialSize; i++)
		{
			AActor* Spawned = Spawn(InClass, Init);
			SpawnedArray.Add(Spawned);
		}
		
		// 생성되서 사용중인 액터 목록으로 들어간 상태이므로
		// 반환처리
		for (auto SpawnedActor : SpawnedArray)
		{
			ReturnPool(SpawnedActor);
		}
	}
}
```


### Deinitialize
데이터 에셋을 통해서 풀을 생성하고 warmup 하는 것도 중요하지만, 적절하게 삭제하는 것 또한 중요하다. 레벨이 바뀌거나 하는 경우에 사용하지 않는 풀들을 삭제해서 자원 사용을 줄여보자.

```cpp file:ObjectPoolSubSystem.cpp
void UObjectPoolSubsystem::Deinitialize()
{
	ClearAllPools();
	Super::Deinitialize();
}

void UObjectPoolSubsystem::ClearAllPools()
{
	for (auto& [Key, _] : PoolMap)
	{
		ClearPool(Key);
	}
	PoolMap.Empty();
}

// 해당 타입의 풀을 삭제하는 함수
void UObjectPoolSubsystem::ClearPool(TSubclassOf<AActor> InClass)
{
	if (!InClass) return;
	if (FActorPool* Pool = PoolMap.Find(InClass))
	{
		// 대기중인 액터들 삭제
		for (AActor* Actor : Pool->ReadyActors)
		{
			if (IsValid(Actor)) Actor->Destroy();
		}
		
		// ReadyActor 지우기
		Pool->ReadyActors.Empty();
		
		// 사용중인 액터들 삭제
		for (auto& [Key,_] : Pool->ActiveActors)
		{
			if (IsValid(Key)) Key->Destroy();
		}
		// ActiveActors 지우기
		Pool->ActiveActors.Empty();
		
		// 해당 타입의 풀을 맵에서 지우기
		// 원래 여기서 문제가 발생해야하는데, ClearAllPools에서 반복자를 통해서 돌고 있으니까는.....
		PoolMap.Remove(InClass);
	}
}
```

# 4. 기타 팁

	```cpp file:UENUM_Example
	UENUM(BlueprintType)

		ReUseOldest		UMETA(ToolTip = "가장 오래된 액터 재사용")
}};
	```
