# 2. GAS Ability (메타 어트리뷰트 연동)
CommitCheck에서 쿨다운과 코스트를 체크 하는데, 메타 어트리뷰트를 사용하는 경우에 제대로된 체크를 할 수 없다. 그러면 어떻게 하나요? -> 체크 코스트를 오버라이드 해서 쓰십시오.... 

```cpp file:Gameablility_Blink.cpp
bool UGameplayAbility_Blink::CheckCost(const FGameplayAbilitySpecHandle Handle, const FGameplayAbilityActorInfo* ActorInfo, OUT FGameplayTagContainer* OptionalRelevantTags) const
{
	UGameplayEffect* CostGE = GetCostGameplayEffect();
	if (!CostGE) return true;

	UAbilitySystemComponent* ASC = ActorInfo ? ActorInfo->AbilitySystemComponent.Get() : nullptr;
	if (!ASC) return false;

	if (!ASC->HasAttributeSetForAttribute(UStatAttributeSet::GetStaminaAttribute())
		|| !ASC->HasAttributeSetForAttribute(UStatAttributeSet::GetStaminaCostAttribute())) return false;

	// 시전자의 현재 스테미너 값
	const float CurrentStamina = ASC->GetNumericAttribute(UStatAttributeSet::GetStaminaAttribute());

	// 필요 소모량을 알기 위해서 Spec 생성
	const FGameplayEffectContextHandle ContextHandle = MakeEffectContext(Handle, ActorInfo);
	const float AbilityLevel = GetAbilityLevel(Handle, ActorInfo);
	FGameplayEffectSpec Spec(CostGE, ContextHandle, AbilityLevel);
	Spec.CalculateModifierMagnitudes();

	// GE에서 StaminaCost 추출
	float StaminaCost = .0f;
	bool bFoundStaminaModifier = false;
	for (int32 ModIndex = 0; ModIndex < Spec.Modifiers.Num(); ModIndex++)
	{
		if (Spec.Def && Spec.Def->Modifiers.IsValidIndex(ModIndex))
		{
			const FGameplayModifierInfo& ModDef = Spec.Def->Modifiers[ModIndex];
			if (ModDef.Attribute == UStatAttributeSet::GetStaminaCostAttribute())
			{
				const FModifierSpec& ModSpec = Spec.Modifiers[ModIndex];
				StaminaCost += ModSpec.GetEvaluatedMagnitude();
				bFoundStaminaModifier = true;
			}
		}
	}

	// GE에 StaminaCost가 없으면 기본 로직
	if (!bFoundStaminaModifier)
	{
		return Super::CheckCost(Handle, ActorInfo, OptionalRelevantTags);
	}

	if (StaminaCost <= CurrentStamina)
	{
		return true;
	}

	// 스테미너가 없어서 실패했다고 CostTag 추가
	const FGameplayTag& CostTag = UAbilitySystemGlobals::Get().ActivateFailCostTag;
	if (OptionalRelevantTags && CostTag.IsValid())
	{
		OptionalRelevantTags->AddTag(CostTag);
	}

	return false;
}
```

```cpp file:GameplayAbility_Sprint
if (bToggleMode)

		// 비동기 태스크, 키 입력되면 델리게이트에 추가된 함수가 실행
		UAbilityTask_WaitInputPress* WaitPressTask = UAbilityTask_WaitInputPress::WaitInputPress(this);
		if (WaitPressTask)
```

```cpp file:TestPlayerCharacter.cpp hlt:7,15
//AbilityLocalInputPresse/Release를 통해서  WaitInputPress/Release Task가 이벤트를 받도록 할 수 있다.
//함수를 호출할 때 ID를 넘기는데이는 GiveAbility시 스펙에 넣어준 ID와 매칭되어 해당 어빌리티로 이어진다.
void ATestPlayerCharacter::OnSprintInputStart()
{
	if (AbilitySystemComponent)
	{
		AbilitySystemComponent->AbilityLocalInputPressed(SprintInputID);
	}
}

void ATestPlayerCharacter::OnSprintInputCompleted()
{
	if (AbilitySystemComponent)
	{
		AbilitySystemComponent->AbilityLocalInputReleased(SprintInputID);
	}
}
```
