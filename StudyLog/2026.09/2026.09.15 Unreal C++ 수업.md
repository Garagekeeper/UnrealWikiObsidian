
# 1.  GAS(Game Ability System)의 개념
게임의 스킬 능력과 관련한 플러그인 포트나이트에서 사용하던걸 플러그인으로 도입 버프/디버프 스탯등을 편리하게. 데이터 기반이기 때문에 편하고 멀티플레이도 완벽 지원.
단점이라면 배울게 좀 많다?

게임플레이 태그를 사용해서 데이터를 사용. 계층적으로 데이터를 사용가능
Game.Buff.Heal 이런식인데 문자열이라 실수하기가 쉬움

- Ability System Component (ASC) (GAS를 관리하고 구동)
	- Admin
	- 편리한 디버깅을 위해서는 Pawn 이상부터 넣기
- Game Play Ability(Gameplay Ability)
	- Skill
- AttributeSet
	- Character Attribute (Status)
	- Effect와 엮임
- GamePlay Effect
	- Buff/Debuff
- GamePlay Tag
	- 식별하기 위한 용도 (주로 상태나 조건)
- GamePlay Cue
	- 시청각 효과

#include "AbilitySystemInterface.h"
#include "TestCharacter.generated.h"

class UStatAttributeSet;
class UTestStatWidget;

UCLASS()
class GAS_STUDY_API ATestCharacter : public ACharacter, public IAbilitySystemInterface
{
	GENERATED_BODY()

public:
	// Sets default values for this character's properties
	ATestCharacter();
	virtual UAbilitySystemComponent* GetAbilitySystemComponent() const override { return AbilitySystemComponent; }

	UStatAttributeSet* GetStatAttributeSet() const;
protected:
	// Called when the game starts or when spawned
	virtual void BeginPlay() override;
	virtual void EndPlay(const EEndPlayReason::Type EndPlayReason) override;

	// Called every frame
	virtual void Tick(float DeltaTime) override;

	// Called to bind functionality to input
	virtual void SetupPlayerInputComponent(class UInputComponent* PlayerInputComponent) override;

	virtual void OnHealthChanged(const FOnAttributeChangeData& InData);

	virtual void OnSteminaChanged(const FOnAttributeChangeData& InData);

protected:
	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "GAS")
	TObjectPtr<UAbilitySystemComponent> AbilitySystemComponent;

	UPROPERTY(VisibleAnywhere, BlueprintReadOnly, Category = "GAS")
	TObjectPtr<UStatAttributeSet> StatAttributeSet;

#include "Test/TestBaseActor.h"

// Sets default values
ATestBaseActor::ATestBaseActor()
{
 	// Set this actor to call Tick() every frame.  You can turn this off to improve performance if you don't need it.
	PrimaryActorTick.bCanEverTick = true;

	AbilitySystemComponent = CreateDefaultSubobject<UAbilitySystemComponent>(TEXT("ASC"));
	// 이렇게 만들어도 에디터상에서 보이지 않는다.
	// 별도의 디버그 방법이 있음
	StatAttributeStat = CreateDefaultSubobject<UStatAttributeSet>(TEXT("Stat"));

	GetMesh()->SetRelativeLocation(FVector(.0f, .0f, -88.f));
	GetMesh()->SetRelativeRotation(FRotator(.0f, -90.f, 0.f));
}

// Called when the game starts or when spawned
void ATestBaseActor::BeginPlay()
{
	Super::BeginPlay();
	
	if (IsValid(AbilitySystemComponent))
	{
		// 이 초기화 이후에 ASC에 바인딩을 해야 유효함
		// OwnerActor : 이 컴포넌트를 가지고 있는 액터 (랜타디의 플레이어)
		// AvatarActor : 이 컴포넌트에 의해 영향을 받는(물리적인) 액터 (랜타디의 타워)
		// 네트워크 파트에서 좀 더 자세한 내용을 다룬다.
		AbilitySystemComponent->InitAbilityActorInfo(this, this);
		
		// 값이 변화했을 때 호출될 함수 바인딩
		FOnGameplayAttributeValueChange& OnHealthChange = AbilitySystemComponent->GetGameplayAttributeValueChangeDelegate(UStatAttributeSet::GetHealthAttribute());
		
	FOnGameplayAttributeValueChange& OnSteminaChange = AbilitySystemComponent->GetGameplayAttributeValueChangeDelegate(UStatAttributeSet::GetSteminaAttribute());
	
	OnHealthChange.AddUObject(this, &ATestCharacter::OnHealthChanged);
	OnSteminaChange.AddUObject(this, &ATestCharacter::OnSteminaChanged);
	}
}

// Called every frame
void ATestBaseActor::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
}

void ATestCharacter::OnHealthChanged(const FOnAttributeChangeData& InData)
{
	if (IsValid(StatWidget) && IsValid(StatAttributeSet))
	{
		StatWidget->SetHealth(InData.NewValue, StatAttributeSet->GetMaxHealth());
	}
}

void ATestCharacter::OnSteminaChanged(const FOnAttributeChangeData& InData)
{
	if (IsValid(StatWidget) && IsValid(StatAttributeSet))
	{
		StatWidget->SetStemina(InData.NewValue, StatAttributeSet->GetMaxStemina());
	}
}

#pragma once

#include "CoreMinimal.h"

/**
 * 
 */
UCLASS()
class GAS_STUDY_API UStatAttributeSet : public UAttributeSet
{
	GENERATED_BODY()
	UStatAttributeSet();
	
public:
	UPROPERTY(BlueprintReadOnly, Category = "Base Stat")
	FGameplayAttributeData Health;
	// 이 매크로를 통해서 프로퍼티 게터, 벨류 세터/게터/이니셜라이져 만들어짐
	ATTRIBUTE_ACCESSORS_BASIC(UStatAttributeSet, Health);

	UPROPERTY(BlueprintReadOnly, Category = "Base Stat")
	FGameplayAttributeData Stamina;
	// 이 매크로를 통해서 프로퍼티 게터, 벨류 세터/게터/이니셜라이져 만들어짐
	ATTRIBUTE_ACCESSORS_BASIC(UStatAttributeSet, Stamina);

	/*
	* #define ATTRIBUTE_ACCESSORS_BASIC(ClassName, PropertyName) \
	GAMEPLAYATTRIBUTE_PROPERTY_GETTER(ClassName, PropertyName) \
	GAMEPLAYATTRIBUTE_VALUE_GETTER(PropertyName) \
	GAMEPLAYATTRIBUTE_VALUE_SETTER(PropertyName) \
	GAMEPLAYATTRIBUTE_VALUE_INITTER(PropertyName)

	이걸 거치면 아래의 함수들이 만들어짐. Set은 앵간하면 건들지 말자
	보통은 이펙트를 통해서 값을 변경하는게 정석
	InitHealth();
	GetHealth();
	GetHealthAttribute();
	InitHealth();
	SetHealth(); (직접적인 사용은 비권장)
	*/
};
